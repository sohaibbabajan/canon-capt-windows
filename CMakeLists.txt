cmake_minimum_required(VERSION 3.20)
project(capt_arm64 VERSION 0.3.1 LANGUAGES CXX)

option(CAPT_ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(CAPT_ENABLE_REFERENCE_TESTS "Compare the codec with the unchanged local C reference" OFF)
include(CTest)

add_library(capt_core STATIC src/protocol.cpp src/raster.cpp src/hiscoa.cpp
    src/control.cpp src/status.cpp src/session.cpp src/device_id.cpp printing/lpr.cpp)
target_include_directories(capt_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(capt_core PUBLIC cxx_std_17)
set(capt_targets capt_core)

if(BUILD_TESTING)
    foreach(component protocol raster hiscoa status session device_id lpr)
        add_executable(test_${component} tests/test_${component}.cpp)
        target_link_libraries(test_${component} PRIVATE capt_core)
        add_test(NAME ${component} COMMAND test_${component})
        list(APPEND capt_targets test_${component})
    endforeach()
    if(CAPT_ENABLE_REFERENCE_TESTS)
        enable_language(C)
        add_library(capt_reference STATIC
            captdriver/src/hiscoa-compress.c tests/reference_hiscoa.c)
        set_target_properties(capt_reference PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
        target_include_directories(capt_reference PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/captdriver/src")
        target_link_libraries(test_hiscoa PRIVATE capt_reference)
        target_compile_definitions(test_hiscoa PRIVATE CAPT_HAS_REFERENCE=1)
        if(CAPT_ENABLE_SANITIZERS)
            target_compile_options(capt_reference PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        endif()
    endif()
endif()

if(WIN32)
    add_subdirectory(windows)
    list(APPEND capt_targets capt_connection_check)
    if(TARGET CaptPrint)
        list(APPEND capt_targets CaptPrint CaptQueue CaptSetup)
    endif()
    if(BUILD_TESTING)
        list(APPEND capt_targets test_windows_worker)
    endif()
endif()

if(CAPT_ENABLE_SANITIZERS AND (MSVC OR NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang"))
    message(FATAL_ERROR "This sanitizer configuration requires GCC or Clang with the GNU command-line interface")
endif()

foreach(target IN LISTS capt_targets)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    if(MINGW)
        target_link_options(${target} PRIVATE -static)
    endif()
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
    elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE
            -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror)
    endif()
    if(CAPT_ENABLE_SANITIZERS)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
endforeach()
