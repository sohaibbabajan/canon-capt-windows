enable_language(RC)
set_source_files_properties(app.rc PROPERTIES OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/app.manifest")
add_executable(capt_connection_check WIN32 main.cpp probe.cpp worker.cpp app.rc)
target_link_libraries(capt_connection_check PRIVATE capt_core cfgmgr32 setupapi user32 gdi32 comdlg32)
target_compile_definitions(capt_connection_check PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
if(MINGW)
    # One executable, using inbox UCRT/Win32 DLLs. No emulated runtime DLLs.
    target_link_options(capt_connection_check PRIVATE -municode -static)
endif()
if(MSVC)
    target_link_options(capt_connection_check PRIVATE /MANIFEST:NO)
endif()

if(BUILD_TESTING)
    add_executable(test_windows_worker ../tests/test_windows_worker.cpp worker.cpp)
    target_link_libraries(test_windows_worker PRIVATE capt_core)
    target_include_directories(test_windows_worker PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_compile_features(test_windows_worker PRIVATE cxx_std_17)
    target_compile_definitions(test_windows_worker PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
    if(MINGW)
        target_link_options(test_windows_worker PRIVATE -municode)
    endif()
    add_test(NAME windows_worker COMMAND test_windows_worker "$<TARGET_FILE:capt_connection_check>")
    set_tests_properties(windows_worker PROPERTIES TIMEOUT 15)
endif()

set(CAPT_PDFIUM "" CACHE PATH "Extracted native ARM64 PDFium distribution")
if(CAPT_PDFIUM)
    set_source_files_properties(print_test.rc PROPERTIES OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/print_test.manifest")
    set_source_files_properties(setup.rc PROPERTIES OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/setup.manifest")
    set(print_sources queue_host.cpp probe.cpp worker.cpp ../printing/document.cpp)
    add_executable(CaptPrint WIN32 ${print_sources} print_test.rc)
    add_executable(CaptQueue WIN32 ${print_sources} queue_service.cpp)
    add_executable(CaptSetup WIN32 setup.cpp worker.cpp setup.rc)
    target_compile_definitions(CaptQueue PRIVATE CAPT_QUEUE)
    target_link_libraries(CaptQueue PRIVATE ws2_32 iphlpapi ole32 advapi32)
    target_link_libraries(CaptSetup PRIVATE capt_core shell32 user32 comctl32 winspool advapi32)
    if(MSVC)
        target_compile_options(CaptSetup PRIVATE /utf-8)
    endif()
    foreach(target CaptPrint CaptQueue)
        target_include_directories(${target} SYSTEM PRIVATE "${CAPT_PDFIUM}/include")
        target_link_libraries(${target} PRIVATE capt_core "${CAPT_PDFIUM}/lib/pdfium.dll.lib" cfgmgr32 setupapi user32)
    endforeach()
    foreach(target CaptPrint CaptQueue CaptSetup)
        target_compile_features(${target} PRIVATE cxx_std_17)
        target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
        if(MINGW)
            target_link_options(${target} PRIVATE -municode)
        endif()
        if(MSVC AND NOT target STREQUAL "CaptQueue")
            target_link_options(${target} PRIVATE /MANIFEST:NO)
        endif()
    endforeach()
endif()
